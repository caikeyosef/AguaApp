    let audioCtx = null;

    function ensureAudio() {
      if (!state.soundEnabled) return null;
      try {
        const Ctx = window.AudioContext || window.webkitAudioContext;
        if (!Ctx) return null;
        if (!audioCtx) audioCtx = new Ctx();
        if (audioCtx.state === 'suspended') audioCtx.resume();
        return audioCtx;
      } catch {
        return null;
      }
    }

    function playGulpSound(beatIndex = 0) {
      const ctx = ensureAudio();
      if (!ctx) return;
      try {
        const now = ctx.currentTime;
        const baseFreq = 210 + beatIndex * 42;

        // Som cômico de "glug" (bolha grave subindo rapidamente)
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(baseFreq, now);
        osc.frequency.exponentialRampToValueAtTime(baseFreq * 2.45, now + 0.085);
        osc.frequency.exponentialRampToValueAtTime(baseFreq * 1.15, now + 0.16);

        gain.gain.setValueAtTime(0.001, now);
        gain.gain.linearRampToValueAtTime(0.18, now + 0.02);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.17);

        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start(now);
        osc.stop(now + 0.18);
      } catch {
        // Ignora bloqueio de autoplay antes da interação do usuário
      }
    }

    function playAhhhSound() {
      const ctx = ensureAudio();
      if (!ctx) return;
      try {
        const now = ctx.currentTime;
        // Acorde cristalino refrescante ("Aaaah!")
        const notes = [523.25, 659.25, 783.99, 1046.5];
        notes.forEach((freq, idx) => {
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = 'triangle';
          const t = now + idx * 0.045;
          osc.frequency.setValueAtTime(freq, t);
          gain.gain.setValueAtTime(0.001, t);
          gain.gain.linearRampToValueAtTime(0.075, t + 0.02);
          gain.gain.exponentialRampToValueAtTime(0.0008, t + 0.36);

          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.start(t);
          osc.stop(t + 0.38);
        });
      } catch {
        // Silencioso
      }
    }

    function playSwitchSound() {
      const ctx = ensureAudio();
      if (!ctx) return;
      try {
        const now = ctx.currentTime;
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(420, now);
        osc.frequency.exponentialRampToValueAtTime(640, now + 0.07);
        gain.gain.setValueAtTime(0.06, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.08);
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start(now);
        osc.stop(now + 0.09);
      } catch {
        // Silencioso
      }
    }

    function clayMat(hex, roughness = 0.50, metalness = 0.04, emissive = 0x000000, emissiveIntensity = 0) {
      const key = `${hex}_${roughness}_${metalness}_${emissive}_${emissiveIntensity}`;
      if (matCache.has(key)) return matCache.get(key);
      const mat = new THREE.MeshStandardMaterial({
        color: hex,
        roughness,
        metalness,
        emissive,
        emissiveIntensity,
        transparent: false,
        opacity: 1.0,
      });
      matCache.set(key, mat);
      return mat;
    }

          mesh,
          vel: new THREE.Vector3(
            Math.cos(angle) * speed,
            isBubble ? 0.9 + Math.random() * 0.8 : isSparkle ? 1.2 + Math.random() * 1.2 : 0.7 + Math.random() * 1.0,
            Math.sin(angle) * speed
          ),
          gravity: isBubble ? -0.45 : isSparkle ? 1.2 : 3.8,
          life: 0,
          maxLife: isBubble ? 1.25 : isSparkle ? 0.95 : 0.65,
        });
      }
    }

    function updateParticles(dt) {
      for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.life += dt;
        if (p.life >= p.maxLife) {
          fxGroup.remove(p.mesh);
          p.mesh.material.dispose();
          particles.splice(i, 1);
          continue;
        }
        p.vel.y -= p.gravity * dt;
        p.mesh.position.addScaledVector(p.vel, dt);
        const t = p.life / p.maxLife;
        p.mesh.material.opacity = Math.max(0, 1 - t);
        p.mesh.scale.multiplyScalar(1 - dt * 0.5);
      }
    }

    // Vetores temporários reutilizados no loop 60fps
    const tmpSpout = new THREE.Vector3();
    const tmpMouth = new THREE.Vector3();
    const tmpMid = new THREE.Vector3();
    const colorOrange = new THREE.Color(0xf97316);
    const colorOrangeEm = new THREE.Color(0xea580c);
    const colorGold = new THREE.Color(0xfacc15);
    const colorGoldEm = new THREE.Color(0xf59e0b);
    const colorAqua = new THREE.Color(0x38bdf8);
    const colorAquaEm = new THREE.Color(0x0ea5e9);
    const colorCyan = new THREE.Color(0xbae6fd);

    function smoothstep(edge0, edge1, x) {
      const t = THREE.MathUtils.clamp((x - edge0) / (edge1 - edge0), 0, 1);
      return t * t * (3 - 2 * t);
    }

    function updateCharacterAnimation(timeSec, dt) {
      if (!activeRig) return;
      const rig = activeRig;
      const rest = rig.idleRest;
      const sp = rig.special;

      // 1. Animação Contínua de Repouso (Respiração, Flutuação, Redemoinho, Chamas)
      const breathe = Math.sin(timeSec * 2.4);
      let scaleY = 1 + breathe * 0.022;
      let scaleXZ = 1 - breathe * 0.012;
      let bodyOffsetY = sp.isFloater ? Math.sin(timeSec * 2.1) * 0.07 : 0;

      if (sp.saciWhirlwind) {
        sp.saciWhirlwind.rotation.y += dt * (state.isDrinking ? 4.8 : 2.2);
        bodyOffsetY = Math.sin(timeSec * 3.2) * 0.045;
      }

      if (sp.mulaFlames) {
        sp.mulaFlames.forEach((f, idx) => {
          f.scale.y = 0.92 + Math.sin(timeSec * (8 + idx * 2) + idx) * 0.14;
          f.rotation.y = Math.sin(timeSec * 4 + idx) * 0.25;
        });
      }

      if (sp.dragonWings) {
        const flap = Math.sin(timeSec * (state.isDrinking ? 14 : 4)) * 0.22;
        sp.dragonWings[0].rotation.z = 0.35 + flap;
        sp.dragonWings[1].rotation.z = -0.35 - flap;
      }

      // 2. Estado Bebendo Água (3 Fases Expressivas)
      if (state.isDrinking) {
        state.drinkTime += dt;
        const t = state.drinkTime;

        // Fase 1 (0.0 -> 0.65s): Levanta recipiente / abre tampa / aproxima jarra
        const raise = smoothstep(0.0, 0.62, t);
        // Fase 2 (0.62 -> 2.38s): 3 Goles rítmicos ("glug-glug-glug")
        const inGulps = t >= 0.62 && t <= 2.38;
        const gulpProgress = THREE.MathUtils.clamp((t - 0.62) / 1.76, 0, 1);
        const gulpPulse = inGulps ? Math.max(0, Math.sin(gulpProgress * Math.PI * 6)) : 0;
        // Fase 3 (2.38 -> 3.35s): Abaixa + Pulinho feliz "Aaaah!"
        const lower = smoothstep(2.35, 2.85, t);
        const activeHold = raise * (1 - lower);

        // Dispara som + gotas nos 3 tempos de gole
        const beatTimes = [0.76, 1.32, 1.88];
        beatTimes.forEach((bt, idx) => {
          if (t >= bt && !state.gulpBeatPlayed[idx]) {
            state.gulpBeatPlayed[idx] = true;
            playGulpSound(idx);
            rig.mouthMarker.getWorldPosition(tmpMouth);
            spawnSplash(tmpMouth, 5, 'drop');
          }
        });

        // Dispara satisfação "Aaaah!" no início da Fase 3
        if (t >= 2.46 && !state.ahhhPlayed) {
          state.ahhhPlayed = true;
          playAhhhSound();
          const meta = getMeta(rig.id);
          const quote = meta.drinkQuotes[state.quoteIdx % meta.drinkQuotes.length];
          state.quoteIdx++;
          setSpeech(quote);

          rig.mouthMarker.getWorldPosition(tmpMouth);
          spawnSplash(tmpMouth, 14, 'sparkle');
          if (sp.isDragon) {
            // Dragão Bebê solta bolhinhas azuis pela boca!
            spawnSplash(tmpMouth, 10, 'bubble');
          }
        }

        // Squash & Stretch da Fase 3 (pulinho feliz!)
        let hopY = 0;
        if (t > 2.42 && t < 3.18) {
          const hp = (t - 2.42) / 0.76;
          hopY = Math.sin(hp * Math.PI) * 0.24;
          const sq = Math.sin(hp * Math.PI * 2) * 0.09;
          scaleY += sq;
          scaleXZ -= sq * 0.6;
        }

        // Pulso da barriga durante os goles
        if (inGulps) {
          scaleXZ += gulpPulse * 0.065;
          scaleY -= gulpPulse * 0.03;
        }

        // Movimentos específicos por tipo de personagem
        if (sp.isMimic) {
          // Mímico abre a tampa inteira e mastiga durante os goles!
          const chomp = inGulps ? Math.sin(gulpProgress * Math.PI * 6) * 0.22 : 0;
          rig.headGroup.rotation.x = rest.headRot.x - activeHold * 0.88 + chomp;
          // Jarra flutuante vai para cima da boca e inclina
          rig.armGroup.position.set(
            THREE.MathUtils.lerp(rest.armPos.x, 0.26, activeHold),
            THREE.MathUtils.lerp(rest.armPos.y, 1.48, activeHold),
            THREE.MathUtils.lerp(rest.armPos.z, 0.22, activeHold)
          );
          rig.cupGroup.rotation.z = activeHold * 1.18;
        } else if (sp.isMula) {
          // Jarra flutuante vai sobre o pescoço e despeja água; chama muda para azul-água!
          rig.armGroup.position.set(
            THREE.MathUtils.lerp(rest.armPos.x, 0.30, activeHold),
            THREE.MathUtils.lerp(rest.armPos.y, 1.98, activeHold),
            THREE.MathUtils.lerp(rest.armPos.z, 0.44, activeHold)
          );
          rig.cupGroup.rotation.z = activeHold * 1.25;

          const aquaBlend = smoothstep(0.65, 1.15, t) * (1 - smoothstep(2.65, 3.25, t));
          sp.mulaFlameMats[0].color.copy(colorOrange).lerp(colorAqua, aquaBlend);
          sp.mulaFlameMats[0].emissive.copy(colorOrangeEm).lerp(colorAquaEm, aquaBlend);
          sp.mulaFlameMats[1].color.copy(colorGold).lerp(colorCyan, aquaBlend);
          sp.mulaFlameMats[1].emissive.copy(colorGoldEm).lerp(colorAqua, aquaBlend);
        } else if (sp.isNessie) {
          // Nessie inclina o pescoço comprido até a tigela e o gole percorre o pescoço!
          rig.headGroup.rotation.x = rest.headRot.x + activeHold * 0.62 - gulpPulse * 0.08;
          if (inGulps && sp.nessieGulp && sp.nessieCurve) {
            sp.nessieGulp.visible = true;
            const singleBeat = (gulpProgress * 3) % 1;
            const pt = sp.nessieCurve.getPoint(THREE.MathUtils.clamp(1 - singleBeat, 0.05, 0.98));
            sp.nessieGulp.position.copy(pt);
            const s = 0.75 + Math.sin(singleBeat * Math.PI) * 0.45;
            sp.nessieGulp.scale.setScalar(s);
          } else if (sp.nessieGulp) {
            sp.nessieGulp.visible = false;
          }
        } else if (sp.isKraken) {
          // 3 Tentáculos com copinhos sobem alternadamente até a boca!
          rig.headGroup.rotation.x = rest.headRot.x - activeHold * 0.24 - gulpPulse * 0.08;
          sp.krakenArms.forEach((ka, idx) => {
            const wave = inGulps ? Math.sin(gulpProgress * Math.PI * 6 + idx * 1.8) * 0.18 : 0;
            ka.pivot.rotation.x = -activeHold * 0.55 + wave;
            ka.cup.rotation.x = -activeHold * 0.55;
          });
        } else {
          // Padrão expressivo para os demais personagens com braço + copo
          const jitterX = sp.isJittery && inGulps ? Math.sin(timeSec * 48) * 0.04 : 0;
          const jitterZ = sp.isJittery && inGulps ? Math.cos(timeSec * 52) * 0.03 : 0;

          if (sp.isStraw) {
            // Cogumelo e ET sugam pelo canudo (inclinam levemente pra frente)
            rig.headGroup.rotation.x = rest.headRot.x + activeHold * 0.14 + gulpPulse * 0.07;
            rig.armGroup.position.set(
              rest.armPos.x - activeHold * 0.16,
              rest.armPos.y + activeHold * 0.24,
              rest.armPos.z + activeHold * 0.10
            );
            rig.armGroup.rotation.x = rest.armRot.x - activeHold * 0.22;
            rig.cupGroup.rotation.z = activeHold * 0.18;
          } else {
            // Virar caneca/cálice/cabaça/balde inclinando a cabeça pra trás!
            rig.headGroup.rotation.x = rest.headRot.x - activeHold * 0.36 - gulpPulse * 0.10;
            rig.headGroup.rotation.z = jitterZ;

            rig.armGroup.position.set(
              rest.armPos.x - activeHold * 0.22 + jitterX,
              rest.armPos.y + activeHold * 0.34,
              rest.armPos.z + activeHold * 0.14
            );
            rig.armGroup.rotation.x = rest.armRot.x - activeHold * 0.65;
            rig.armGroup.rotation.z = rest.armRot.z + activeHold * 0.25;
            rig.cupGroup.rotation.z = activeHold * 0.92;
          }

          // Sapo Mago infla a garganta/papo em cada gole!
          if (sp.frogThroat) {
            const puff = inGulps ? 1 + gulpPulse * 0.52 : 1;
            sp.frogThroat.scale.set(1.15 * puff, 0.75 * puff, 0.85 * puff);
          }
        }

        // Jato animado de água azul (Phase 2)
        if (inGulps) {
          rig.spoutMarker.getWorldPosition(tmpSpout);
          rig.mouthMarker.getWorldPosition(tmpMouth);
          tmpMid.addVectors(tmpSpout, tmpMouth).multiplyScalar(0.5);
          tmpMid.y += sp.isStraw ? 0.04 : 0.14;

          streamBeads.forEach((bead, i) => {
            bead.visible = true;
            const u = ((timeSec * 3.8 + i / STREAM_COUNT) % 1);
            // Curva quadrática de Bézier entre o bico do copo e a boca
            const inv = 1 - u;
            bead.position.set(
              inv * inv * tmpSpout.x + 2 * inv * u * tmpMid.x + u * u * tmpMouth.x,
              inv * inv * tmpSpout.y + 2 * inv * u * tmpMid.y + u * u * tmpMouth.y,
              inv * inv * tmpSpout.z + 2 * inv * u * tmpMid.z + u * u * tmpMouth.z
            );
            const pulseSize = 0.7 + Math.sin(u * Math.PI) * 0.55;
            bead.scale.setScalar(pulseSize);
          });
        } else {
          streamBeads.forEach((b) => (b.visible = false));
        }

        rig.bodyGroup.position.y = rest.bodyY + bodyOffsetY + hopY;
        rig.bodyGroup.scale.set(scaleXZ, scaleY, scaleXZ);

        // Encerra o ciclo
        if (t >= state.drinkDuration) {
          stopDrinking(true);
          if (state.loopEnabled) {
            state.loopCooldown = 0.65;
          }
        }
      } else {
        // Estado ocioso suave
        rig.bodyGroup.position.y = rest.bodyY + bodyOffsetY;
        rig.bodyGroup.scale.set(scaleXZ, scaleY, scaleXZ);
        if (!sp.isMimic && !sp.isMula) {
          rig.headGroup.rotation.z = Math.sin(timeSec * 1.4) * 0.04;
        }
        if (state.loopEnabled) {
          state.loopCooldown -= dt;
          if (state.loopCooldown <= 0) {
            triggerDrink();
          }
        }
      }
    }

    const exporter = new GLTFExporter();

    function exportCurrentGLB() {
      if (!activeRig) return;
      const meta = getMeta(activeRig.id);

      // Pausa animação brevemente na pose limpa para exportar o modelo bem alinhado
      const wasDrinking = state.isDrinking;
      stopDrinking(true);
      showToast(`Gerando arquivo 3D de ${meta.name}...`);

      try {
        exporter.parse(
          activeRig.root,
          (result) => {
            if (result instanceof ArrayBuffer) {
              const blob = new Blob([result], { type: 'model/gltf-binary' });
              const url = URL.createObjectURL(blob);
              const link = document.createElement('a');
              link.href = url;
              link.download = `gotaviva-${meta.id}.glb`;
              document.body.appendChild(link);
              link.click();
              document.body.removeChild(link);
              setTimeout(() => URL.revokeObjectURL(url), 1000);
              showToast(`Modelo "${meta.name}.glb" baixado com sucesso!`);
            } else {
              showToast('Erro inesperado ao gerar binário GLB.');
            }
            if (wasDrinking || state.loopEnabled) triggerDrink();
          },
          (err) => {
            console.error('Erro GLTFExporter:', err);
            showToast('Não foi possível exportar o .GLB neste navegador.');
          },
          { binary: true }
        );
      } catch (err) {
        console.error(err);
        showToast('Erro ao exportar modelo 3D.');
      }
    }

      camera.updateProjectionMatrix();

    function animate(now) {
      requestAnimationFrame(animate);
      const dt = Math.min(0.05, (now - lastNow) / 1000);
      lastNow = now;
      const timeSec = now / 1000;

      updateCharacterAnimation(timeSec, dt);
      updateParticles(dt);

      controls.update();
      renderer.render(scene, camera);
    }
